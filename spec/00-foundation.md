├── src/
│   ├── index.ts          # JSII public surface — re-exports only

│   └── {name}-fns.ts     # Pure functions (internal logic, not JSII-exported directly)

    └── {name}-fns.test.ts # Pure function unit tests (no CDK)

## Functional Architecture Pattern

All business logic lives in pure functions. The Construct class is the thin impure edge required by CDK/JSII.

### The split

**Pure functions** (`src/{name}-fns.ts`) — no side effects, no CDK constructs, fully unit-testable:

```typescript
// src/sqs-queue-fns.ts
import { Duration } from 'aws-cdk-lib';
import { QueueProps } from 'aws-cdk-lib/aws-sqs';
import { Context, contextId } from '@sevenpico/cdk-context';

export interface SqsQueueOptions {
  readonly visibilityTimeoutSeconds?: number;
  readonly retentionPeriodSeconds?: number;
  readonly fifo?: boolean;
}

export const sqsQueueProps = (ctx: Context, opts: SqsQueueOptions): QueueProps => ({
  queueName: opts.fifo ? `${contextId(ctx)}.fifo` : contextId(ctx),
  visibilityTimeout: Duration.seconds(opts.visibilityTimeoutSeconds ?? 30),
  retentionPeriod: Duration.seconds(opts.retentionPeriodSeconds ?? 345600),
  fifo: opts.fifo ?? false,
});
```

**Construct class** (`src/{name}.ts`) — CDK glue only, no logic:

```typescript
// src/sqs-queue.ts
import { Construct } from 'constructs';
import { Tags } from 'aws-cdk-lib';
import { Queue } from 'aws-cdk-lib/aws-sqs';
import { Context, contextTags, isEnabled } from '@sevenpico/cdk-context';
import { sqsQueueProps, SqsQueueOptions } from './sqs-queue-fns';

export interface SqsQueueProps extends SqsQueueOptions {
  readonly context: Context;
}

export class SqsQueue extends Construct {
  public readonly queue?: Queue;

  constructor(scope: Construct, id: string, props: SqsQueueProps) {
    super(scope, id);
    if (!isEnabled(props.context)) return;

    this.queue = new Queue(this, 'Queue', sqsQueueProps(props.context, props));
    applyContextTags(this, props.context);
  }
}
```

### Tag application helper

Tags cannot be set via CDK props — they are applied imperatively via `Tags.of()`. Use this pattern in every construct constructor:

```typescript
// shared utility — inline in each constructor
const applyContextTags = (scope: Construct, ctx: Context): void => {
  Object.entries(contextTags(ctx)).forEach(([k, v]) => Tags.of(scope).add(k, v));
};
```

### Partial application for child contexts

When a construct creates multiple sub-resources with context variations, use `extendContext`:

```typescript
import { extendContext } from '@sevenpico/cdk-context';

const dlqCtx = extendContext(props.context, { attributes: ['dlq'] });
// dlqCtx.id → 'ns-env-stage-name-dlq'
```

### `enabled` pattern

If `!isEnabled(props.context)`, return early. All public properties remain `undefined`. Callers must handle optional properties.

```typescript
constructor(scope, id, props) {
  super(scope, id);
  if (!isEnabled(props.context)) return;
  // ... create resources
}
```

## JSII Constraints

JSII imposes strict limits on the public API surface. Follow these rules in all `src/index.ts` and `src/{name}.ts` exports:

| Constraint | Rule |
|-----------|------|
| No function types | Do not use `(x: T) => R` in public interfaces |
| No union types | Use separate optional props instead of `string \| number` |
| No generics | Avoid `Array<T>` in public interfaces; use `string[]` etc. |
| No `any` | Use specific interfaces; `object` as last resort |
| `readonly` everywhere | All interface properties must be `readonly` |
| Export everything | All public types must be re-exported from `index.ts` |
| No default exports | Named exports only |

Pure functions in `{name}-fns.ts` are **internal TypeScript** and not part of the JSII surface — they are exempt from these constraints.

## Testing Conventions

**Pure function tests** (no CDK, fast):
```typescript
// test/sqs-queue-fns.test.ts
import { makeContext } from '@sevenpico/cdk-context';
import { sqsQueueProps } from '../src/sqs-queue-fns';

test('uses contextId as queue name', () => {
  const ctx = makeContext({ namespace: '7p', stage: 'test', name: 'orders' });
  expect(sqsQueueProps(ctx, {}).queueName).toBe('7p-test-orders');
});
```

**Construct integration tests** (CDK assertions):
```typescript
// test/sqs-queue.test.ts
import { App, Stack } from 'aws-cdk-lib';
import { Template } from 'aws-cdk-lib/assertions';
import { makeContext } from '@sevenpico/cdk-context';
import { SqsQueue } from '../src/sqs-queue';

test('creates queue with context name', () => {
  const app = new App();
  const stack = new Stack(app, 'Test');
  new SqsQueue(stack, 'Q', { context: makeContext({ namespace: '7p', stage: 'test', name: 'orders' }) });
  Template.fromStack(stack).hasResourceProperties('AWS::SQS::Queue', { QueueName: '7p-test-orders' });
});

test('creates no resources when disabled', () => {
  const app = new App();
  const stack = new Stack(app, 'Test');
  new SqsQueue(stack, 'Q', { context: makeContext({ namespace: '7p', enabled: false }) });
  Template.fromStack(stack).resourceCountIs('AWS::SQS::Queue', 0);
});
```

## Build Commands

```bash
# Bootstrap monorepo (first time)
npx projen

# Build all packages
npx nx run-many --target=build --all

# Build a single package
npx nx build @sevenpico/cdk-construct-sqs-queue

# Test all
npx nx run-many --target=test --all

# Build in dependency order (automatic via nx)
npx nx affected --target=build
```

## Package Versioning

All packages in this monorepo are versioned together (monorepo-style). Projen's release workflow bumps all packages to the same version on each release. This simplifies inter-package dependency management.
